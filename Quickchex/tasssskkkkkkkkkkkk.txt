    DailyTaskResponse

router = APIRouter()

# --- CREATE TASK ---
@router.post("/", response_model=DailyTaskResponse)
def create_task(data: DailyTaskCreate, db: Session = Depends(get_db)):
    task_data = data.model_dump() if hasattr(data, 'model_dump') else data.dict()
   
    # Automatically fetch designation based on the sheet's employee
    user_info = (
        db.query(Profile.designation)
        .join(MonthlySheet, MonthlySheet.employee_id == Profile.id)
        .filter(MonthlySheet.id == data.monthly_sheet_id)
        .first()
    )
 
    task_data["designation"] = user_info.designation if user_info else "Not Assigned"
 
    # Save to database
    task = DailyTask(**task_data)
    db.add(task)
    db.commit()
    db.refresh(task)
    return task
 
# --- GET SINGLE TASK (For View Modal) ---
@router.get("/{task_id}", response_model=DailyTaskResponse)
def get_task(task_id: int, db: Session = Depends(get_db)):
    task = db.query(DailyTask).filter(DailyTask.id == task_id).first()
    if not task:
        raise HTTPException(status_code=404, detail="Task not found")
    return task
 
# --- GET ALL TASKS FOR A SHEET ---
@router.get("/by-sheet/{sheet_id}", response_model=list[DailyTaskResponse])
def get_tasks_by_sheet(sheet_id: int, db: Session = Depends(get_db)):
    return db.query(DailyTask).filter(DailyTask.monthly_sheet_id == sheet_id).all()
 
# --- UPDATE TASK (The Database Update Logic) ---
@router.put("/{task_id}", response_model=DailyTaskResponse)
def update_task(task_id: int, data: DailyTaskCreate, db: Session = Depends(get_db)):
    # 1. Fetch the existing task from the database
    task = db.query(DailyTask).filter(DailyTask.id == task_id).first()
   
    if not task:
        raise HTTPException(status_code=404, detail="Task not found")
 
    # 2. Extract the update data (handles Pydantic v1 or v2)
    update_data = data.model_dump(exclude_unset=True) if hasattr(data, 'model_dump') else data.dict(exclude_unset=True)
   
    # 3. Map the new data to the existing database record
    for key, value in update_data.items():
        setattr(task, key, value)
 
    # 4. CRITICAL: Commit the transaction to save changes permanently
    db.commit()
   
    # 5. Refresh to get the updated object back from DB
    db.refresh(task)
   
    return task
 
# --- DELETE TASK ---
@router.delete("/{task_id}")
def delete_task(task_id: int, db: Session = Depends(get_db)):
    task = db.query(DailyTask).filter(DailyTask.id == task_id).first()
    if not task:
        raise HTTPException(status_code=404, detail="Task not found")
   
    db.delete(task)
    db.commit()
    return {"message": "Deleted successfully"}

from sqlalchemy.orm import Session
from fastapi import HTTPException
from app.models.user import User

from app.models.task_model import Task
from app.repositories.task_repo import (
    create_task,
    get_tasks_by_user,
    get_all_tasks,
    get_task_by_id
)


def assign_task_service(db: Session, data, current_user):

    # 🔍 Check assigned user exists
    assigned_user = db.query(User).filter(User.id == data.assigned_to).first()

    if not assigned_user:
        raise HTTPException(status_code=404, detail="Assigned user not found")

    # 🚫 Prevent assigning to admin (optional rule)
    if assigned_user.role.name == "Admin":
        raise HTTPException(status_code=400, detail="Cannot assign task to Admin")

    task = Task(
        title=data.title,
        description=data.description,
        assigned_to=data.assigned_to,
        assigned_by=current_user.id,
    )

    return create_task(db, task)


def my_tasks_service(db: Session, user_id: int):
    return get_tasks_by_user(db, user_id)


def all_tasks_service(db: Session):
    return get_all_tasks(db)


def update_task_status_service(db: Session, task_id: int, status: str, user_id: int):
    task = get_task_by_id(db, task_id)

    if not task:
        raise HTTPException(status_code=404, detail="Task not found")

    if task.assigned_to != user_id:
        raise HTTPException(status_code=403, detail="Not allowed")

    allowed_status = ["Pending", "In Progress", "Done"]

    if status not in allowed_status:
        raise HTTPException(status_code=400, detail="Invalid status")

    task.status = status
    db.commit()
    return task
